cmake_minimum_required(VERSION 3.20)
project(sizetree VERSION 0.1.0 LANGUAGES CXX)

if(NOT UNIX)
  message(FATAL_ERROR "sizetree requires a POSIX terminal (Linux or macOS).")
endif()

option(SIZETREE_STATIC "Build a fully statically linked executable (Linux only)" OFF)
if(SIZETREE_STATIC AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Fully static executables are supported only on Linux.")
endif()

find_package(Threads REQUIRED)
function(configure_sizetree_target target)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE _FILE_OFFSET_BITS=64 SIZETREE_VERSION="${PROJECT_VERSION}")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(${target} PRIVATE stdc++fs)
  endif()
  if(SIZETREE_STATIC)
    set_property(TARGET ${target} APPEND_STRING PROPERTY LINK_FLAGS " -static")
  endif()
endfunction()

add_executable(sizetree src/main.cpp src/cache.cpp src/config.cpp)
configure_sizetree_target(sizetree)

include(GNUInstallDirs)
install(TARGETS sizetree RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES LICENSE DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/sizetree")
install(FILES README.md CHANGELOG.md DESTINATION "${CMAKE_INSTALL_DOCDIR}")

include(CTest)
if(BUILD_TESTING)
  add_executable(sizetree_refresh_tests tests/test_periodic_refresh.cpp src/cache.cpp src/config.cpp)
  configure_sizetree_target(sizetree_refresh_tests)
  add_test(NAME sizetree_periodic_refresh COMMAND sizetree_refresh_tests)
  set_tests_properties(sizetree_periodic_refresh PROPERTIES TIMEOUT 15)
  find_package(Python3 3.6 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME sizetree_cli
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_sizetree.py" "$<TARGET_FILE:sizetree>")
    set_tests_properties(sizetree_cli PROPERTIES TIMEOUT 30)
  endif()
endif()
